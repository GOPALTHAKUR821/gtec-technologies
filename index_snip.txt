function getCart(){
  try{
    const raw=JSON.parse(localStorage.getItem('gtec_cart')||'[]');
    return Array.isArray(raw)?raw.map(x=>({...x,price:Number(String(x.price??0).replace(/[^0-9.]/g,''))||0,qty:Math.max(1,Number(x.qty)||1)})):[]; 
  }catch{return []}
}

function addProductFromButton(btn){
  try{
    if(!btn) throw new Error('Button missing');
    let raw=btn.getAttribute('data-product');
    if(raw){
      const product=JSON.parse(raw);
      addToCart(product);
      return true;
    }
    raw=btn.getAttribute('data-product-js');
    if(raw){
      // HTML attributes are decoded by the browser before getAttribute().
      const product=JSON.parse(raw);
      addToCart(product);
      return true;
    }
    throw new Error('Product data missing');
  }catch(e){
    console.error('Add to cart error:',e,btn);
    alert('Product cart me add nahi ho paya. Dobara try karein.');
    return false;
  }
}
function addToCart(product){
  const price=Number(String(product.price??0).replace(/[^0-9.]/g,''))||0;
  const normalized={...product,price,id:String(product.id),name:String(product.name||'Product')};
  const c=getCart(); const found=c.find(x=>String(x.id)===normalized.id);
  if(found){found.qty++;found.price=price;} else c.push({...normalized,qty:1});
  saveCart(c); renderCheckout();
  alert(normalized.name+' added to cart.');
}
function changeCartQty(id,delta){
  const c=getCart(); const p=c.find(x=>String(x.id)===String(id)); if(!p)return;
  p.qty+=delta;if(p.qty<=0)c.splice(c.indexOf(p),1);saveCart(c);renderCheckout();
}
function removeFromCart(id){saveCart(getCart().filter(x=>String(x.id)!==String(id)));renderCheckout();}
function cartSubtotal(){return getCart().reduce((s,x)=>s+(Number(x.price)||0)*(Number(x.qty)||0),0);}
function estimateDelivery(pin,city){
  pin=String(pin||'').replace(/\D/g,'').slice(0,6);city=String(city||'').toLowerCase().trim();
  if(pin.length!==6)return null;
  if(pin.startsWith('204')||city.includes('hathras'))return 50;
  if(pin.startsWith('202')||pin.startsWith('281')||pin.startsWith('282')||pin.startsWith('283'))return 70;
  if(/^(20|21|22|23|24|25|26|27|28)/.test(pin))return 90;
  if(pin.startsWith('110')||pin.startsWith('121')||pin.startsWith('122')||pin.startsWith('201')||pin.startsWith('203'))return 120;
  if(/^(14|16|17|18|30)/.test(pin))return 150;
  return 200;
}

async function placeCartOrder(e){
  e.preventDefault();const c=getCart();if(!c.length){alert('Cart empty hai.');return;}
  const payment=document.querySelector('input[name="coPayment"]:checked').value,utr=(document.getElementById('coUtr')?.value||'').trim();
  if(payment==='UPI'&&!utr){alert('UPI UTR / Transaction ID enter karein.');return;}
  const pin=document.getElementById('coPin').value.trim(),city=document.getElementById('coCity').value.trim(),delivery=estimateDelivery(pin,city);
  if(delivery===null){alert('Valid 6-digit PIN code enter karein.');return;}
  const payload={name:document.getElementById('coName').value.trim(),phone:document.getElementById('coPhone').value.trim(),address:document.getElementById('coAddress').value.trim(),city,pin,installation:document.getElementById('coInstall').value,payment,utr,items:c.map(x=>({productId:x.id,name:x.name,price:Number(x.price)||0,qty:Number(x.qty)||0}))};
  try{
    const r=await fetch('/api/orders',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(payload)});
    const data=await r.json();if(!r.ok)throw new Error(data.error||'Order failed');
    localStorage.removeItem('gtec_cart');updateCartBar();renderCheckout();
    document.getElementById('coSuccess').style.display='block';
    document.getElementById('coSuccess').textContent='Order '+data.order.id+' placed successfully. Total ₹'+Number(data.order.total).toLocaleString('en-IN')+'.';
    document.getElementById('checkoutForm').reset();document.querySelector('input[name="coPayment"][value="COD"]').checked=true;document.getElementById('coUpiBox').style.display='none';updateCheckoutTotals();
  }catch(err){alert('Order place nahi hua: '+err.message+'. Server/backend running hai ya nahi check karein.');}
}
